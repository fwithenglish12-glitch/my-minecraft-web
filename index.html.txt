    <script>
        // --- 1. KHỞI TẠO GAME & MAP ---
        let currentMode = "Survival"; // Chế độ: Survival, Creative, Spectator
        let playerHp = 10, playerFood = 10, activeSlot = 1;
        let blocks = [], blockMap = new Map();

        const ITEM_COLORS = { 1: 0x5b8731, 2: 0x866043, 3: 0x707070, 4: 0x2c2c2c, 5: 0x151515, 6: 0xa52a2a, 7: 0x00ffff, 8: 0x00008b, 9: 0xffd700 };

        const scene = new THREE.Scene(); scene.background = new THREE.Color(0x78a7ff);
        scene.fog = new THREE.FogExp2(0x78a7ff, 0.03);

        const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
        const renderer = new THREE.WebGLRenderer({ antialias: true });
        renderer.setSize(window.innerWidth, window.innerHeight);
        document.body.appendChild(renderer.domElement);

        const ambientLight = new THREE.AmbientLight(0xffffff, 0.6); scene.add(ambientLight);
        const sunLight = new THREE.DirectionalLight(0xffffff, 0.5); sunLight.position.set(20, 40, 20); scene.add(sunLight);

        // Tạo vật liệu pixel thô sơ
        function createMat(color) {
            const canvas = document.createElement('canvas'); canvas.width = 16; canvas.height = 16;
            const ctx = canvas.getContext('2d'); ctx.fillStyle = '#' + color.toString(16).padStart(6, '0'); ctx.fillRect(0, 0, 16, 16);
            for (let i=0; i<16; i+=4) { for (let j=0; j<16; j+=4) { ctx.fillStyle = Math.random() > 0.5 ? 'rgba(0,0,0,0.05)' : 'rgba(255,255,255,0.05)'; ctx.fillRect(i, j, 4, 4); } }
            const tex = new THREE.CanvasTexture(canvas); tex.magFilter = THREE.NearestFilter;
            return new THREE.MeshStandardMaterial({ map: tex, roughness: 0.9 });
        }

        const mats = {}; for (let k in ITEM_COLORS) mats[k] = createMat(ITEM_COLORS[k]);
        const geom = new THREE.BoxGeometry(1, 1, 1);

        function generateWorld() {
            for (let x = -12; x < 25; x++) {
                for (let z = -12; z < 25; z++) {
                    let h = Math.floor((Math.sin(x * 0.15) + Math.cos(z * 0.15)) * 2) + 4;
                    for (let y = -10; y <= h; y++) {
                        let type = 3;
                        if (y === h) type = 1;
                        else if (y > h - 3) type = 2;
                        else if (y < -5) type = 4;
                        
                        // Đền cổ hành lang sâu dưới đất
                        if (x > 4 && x < 16 && z > 4 && z < 16 && y >= -9 && y <= -6) {
                            if (x === 5 || x === 15 || z === 5 || z === 15 || y === -9) type = 5;
                            else continue; // Để rỗng lòng cho người chơi đi bộ hành lang
                        }
                        if (type === 4 && Math.random() < 0.02) type = 7; // Quặng kim cương

                        spawnBlock(x, y, z, type);
                    }
                }
            }
        }

        function spawnBlock(x, y, z, type) {
            const key = `${x},${y},${z}`; if (blockMap.has(key)) return;
            const b = new THREE.Mesh(geom, mats[type]); b.position.set(x, y, z);
            scene.add(b); b.userData = { type: type, key: key };
            blocks.push(b); blockMap.set(key, b);
        }
        generateWorld();

        // --- 2. VẬT LÝ & ĐIỀU KHIỂN WASD MƯỢT MÀ ---
        let player = { pos: new THREE.Vector3(10, 10, 10), vel: new THREE.Vector3(), height: 1.6, radius: 0.35, grounded: false };
        camera.position.copy(player.pos);

        let keys = {}, yaw = 0, pitch = 0;

        window.addEventListener('keydown', e => {
            keys[e.key.toLowerCase()] = true;
            if (parseInt(e.key) >= 1 && parseInt(e.key) <= 9) {
                activeSlot = parseInt(e.key);
                document.querySelectorAll('.slot').forEach(s => s.classList.remove('selected'));
                document.querySelector(`.slot[data-id="${activeSlot}"]`).classList.add('selected');
            }
            if (e.key.toLowerCase() === 'p') {
                currentMode = currentMode === "Survival" ? "Creative" : (currentMode === "Creative" ? "Spectator" : "Survival");
                document.getElementById('modeTxt').innerText = currentMode;
            }
        });
        window.addEventListener('keyup', e => { keys[e.key.toLowerCase()] = false; });
        window.addEventListener('click', () => { document.body.requestPointerLock(); });

        document.addEventListener('mousemove', e => {
            if (document.pointerLockElement === document.body) {
                yaw -= e.movementX * 0.0025; pitch -= e.movementY * 0.0025;
                pitch = Math.max(-Math.PI/2 + 0.05, Math.min(Math.PI/2 - 0.05, pitch));
                camera.quaternion.setFromEuler(new THREE.Euler(pitch, yaw, 0, 'YXZ'));
            }
        });

        function checkCollide(pPos) {
            if (currentMode === "Spectator") return false;
            let minX = Math.floor(pPos.x - player.radius), maxX = Math.floor(pPos.x + player.radius);
            let minY = Math.floor(pPos.y - player.height), maxY = Math.floor(pPos.y + 0.1);
            let minZ = Math.floor(pPos.z - player.radius), maxZ = Math.floor(pPos.z + player.radius);
            for (let x = minX; x <= maxX; x++) {
                for (let y = minY; y <= maxY; y++) {
                    for (let z = minZ; z <= maxZ; z++) {
                        if (blockMap.has(`${x},${y},${z}`)) return true;
                    }
                }
            }
            return false;
        }

        // --- 3. ĐẬP / ĐẶT BLOCK CHUẨN XÁC ---
        const raycaster = new THREE.Raycaster();
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        function playBeep(f, d) { try { let o = audioCtx.createOscillator(), g = audioCtx.createGain(); o.frequency.value = f; o.connect(g); g.connect(audioCtx.destination); g.gain.setValueAtTime(0.1, audioCtx.currentTime); g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime+d); o.start(); o.stop(audioCtx.currentTime+d); } catch(e){} }

        window.addEventListener('mousedown', e => {
            if (document.pointerLockElement !== document.body || currentMode === "Spectator") return;
            raycaster.setFromCamera(new THREE.Vector2(0,0), camera);
            let intersects = raycaster.intersectObjects(blocks);
            if (intersects.length > 0 && intersects[0].distance < 5.5) {
                let obj = intersects[0].object;
                if (e.button === 0) { // Chuột trái: Phá khối
                    playBeep(120, 0.1); scene.remove(obj);
                    blockMap.delete(obj.userData.key); blocks.splice(blocks.indexOf(obj), 1);
                } else if (e.button === 2) { // Chuột phải: Đặt khối
                    let normal = intersects[0].face.normal; let nPos = obj.position.clone().add(normal);
                    // Tránh đặt khối đè lên người
                    if (Math.floor(nPos.x) === Math.floor(player.pos.x) && Math.floor(nPos.z) === Math.floor(player.pos.z) && (Math.floor(nPos.y) === Math.floor(player.pos.y) || Math.floor(nPos.y) === Math.floor(player.pos.y-1))) return;
                    playBeep(200, 0.08); spawnBlock(nPos.x, nPos.y, nPos.z, activeSlot);
                }
            }
        });
        window.addEventListener('contextmenu', e => e.preventDefault());

        // --- 4. HỆ THỐNG SINH TỒN & TOTEM ---
        function drawHud() {
            let h = '', f = '';
            for(let i=0; i<10; i++) h += `<span class="heart" style="color:${i < playerHp ? '#ff2222' : '#444'}">♥</span>`;
            for(let i=0; i<10; i++) f += `<span class="food" style="color:${i < playerFood ? '#ffaa00' : '#444'}">♦</span>`;
            document.getElementById('heart-bar').innerHTML = h; document.getElementById('food-bar').innerHTML = f;
        }
        drawHud();

        function takeDamage(amt) {
            if (currentMode !== "Survival" || playerHp <= 0) return;
            if (playerHp - amt <= 0) {
                if (activeSlot === 9) { // Cứu mạng bằng Totem ở ô số 9
                    playBeep(450, 0.6); let fx = document.getElementById('totem-fx'), box = document.getElementById('totem-box');
                    fx.style.display = 'flex'; let s = 0.1;
                    let inter = setInterval(() => { s += 0.2; box.style.transform = `scale(${s}) rotate(${s*15}deg)`; if(s>=1.3) { clearInterval(inter); setTimeout(() => fx.style.display='none', 500); } }, 30);
                    playerHp = 4; playerFood = 8; drawHud(); return;
                }
                playerHp = 0; drawHud(); document.getElementById('game-over').style.display = 'flex'; document.exitPointerLock(); return;
            }
            playerHp -= amt; playBeep(100, 0.2); drawHud();
        }

        document.getElementById('respawn-btn').addEventListener('click', () => {
            playerHp = 10; playerFood = 10; player.pos.set(10, 10, 10); player.vel.set(0,0,0); drawHud();
            document.getElementById('game-over').style.display = 'none'; document.body.requestPointerLock();
        });

        // --- 5. GAME LOOP CHẠY LIÊN TỤC MƯỢT MÀ ---
        let clock = new THREE.Clock();
        function animate() {
            requestAnimationFrame(animate);
            let dt = Math.min(clock.getDelta(), 0.1);

            if (playerHp > 0 || currentMode !== "Survival") {
                let speed = currentMode === "Creative" ? 12 : 5.5;
                let f = new THREE.Vector3(0, 0, -1).applyQuaternion(camera.quaternion);
                let s = new THREE.Vector3(1, 0, 0).applyQuaternion(camera.quaternion);
                if (currentMode !== "Spectator") { f.y = 0; s.y = 0; } f.normalize(); s.normalize();

                let move = new THREE.Vector3();
                if (keys['w']) move.add(f); if (keys['s']) move.add(f.clone().negate());
                if (keys['a']) move.add(s.clone().negate()); if (keys['d']) move.add(s);
                move.normalize().multiplyScalar(speed * dt);

                // Trọng lực vật lý trượt tường mượt
                if (currentMode !== "Spectator") {
                    player.vel.y -= 24 * dt;
                    if (keys[' '] && player.grounded) { player.vel.y = 8.0; player.grounded = false; }
                } else {
                    player.vel.y = 0; if (keys[' ']) player.pos.y += speed * dt; if (keys['shift']) player.pos.y -= speed * dt;
                }

                let next = player.pos.clone();
                next.x += move.x; if (!checkCollide(next)) player.pos.x = next.x; else next.x = player.pos.x;
                next.z += move.z; if (!checkCollide(next)) player.pos.z = next.z; else next.z = player.pos.z;

                if (currentMode !== "Spectator") {
                    next.y += player.vel.y * dt;
                    if (!checkCollide(next)) { player.pos.y = next.y; player.grounded = false; }
                    else {
                        if (player.vel.y < -12) { let dmg = Math.floor((Math.abs(player.vel.y) - 10) * 0.5); if (dmg > 0) takeDamage(dmg); }
                        if (player.vel.y < 0) player.grounded = true; player.vel.y = 0;
                    }
                }
            }
            camera.position.copy(player.pos); renderer.render(scene, camera);
        }
        animate();

        window.addEventListener('resize', () => { camera.aspect = window.innerWidth / window.innerHeight; camera.updateProjectionMatrix(); renderer.setSize(window.innerWidth, window.innerHeight); });
    </script>
